%%bash

# 1. Create a sample encrypted passwords file
cat << 'EOF' > encrypted_passwords.txt
uftu
ifmmp
xpsme
EOF

# 2. Write your CUDA code to main.cu
cat << 'EOF' > main.cu
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <cuda_runtime.h>

#define MAX_PW_LEN 256

// CUDA Error Checking Macro
#define cudaCheckError() { \
    cudaError_t e=cudaGetLastError(); \
    if(e!=cudaSuccess) { \
        printf("CUDA Error %s:%d: %s\n", __FILE__, __LINE__, cudaGetErrorString(e)); \
        exit(1); \
    } \
}

// -------------------------------------------------------------------------
// THIS IS A PLACEHOLDER DECRYPTION KERNEL
// Replace the logic inside this kernel with the encryption function 
// provided by your lecturer (converted to decryption).
// -------------------------------------------------------------------------
__global__ void decrypt_kernel(char* d_encrypted, char* d_decrypted, int num_passwords) {
    // Calculate global thread ID in the X dimension
    int idx = blockIdx.x * blockDim.x + threadIdx.x;

    // Ensure thread does not access out of bounds memory
    if (idx < num_passwords) {
        // Pointer to this specific thread's password
        char* enc_pw = d_encrypted + (idx * MAX_PW_LEN);
        char* dec_pw = d_decrypted + (idx * MAX_PW_LEN);

        // --- REPLACE THIS WITH YOUR PROVIDED ALGORITHM ---
        // Example Mock Decryption: A simple Caesar cipher shift (-1)
        int i = 0;
        while (enc_pw[i] != '\0' && enc_pw[i] != '\n' && enc_pw[i] != '\r' && i < MAX_PW_LEN - 1) {
            dec_pw[i] = enc_pw[i] - 1; 
            i++;
        }
        dec_pw[i] = '\0';
        // -------------------------------------------------
    }
}

int main(int argc, char* argv[]) {
    if (argc != 2) {
        printf("Usage: %s <encrypted_passwords_file>\n", argv[0]);
        return 1;
    }

    const char* filename = argv[1];
    
    // --- 1. Read the file to determine the number of passwords ---
    FILE* file = fopen(filename, "r");
    if (!file) {
        printf("Error: Could not open file %s\n", filename);
        return 1;
    }

    int num_passwords = 0;
    char buffer[MAX_PW_LEN];
    while (fgets(buffer, MAX_PW_LEN, file)) {
        num_passwords++;
    }
    
    if (num_passwords == 0) {
        printf("Error: File is empty.\n");
        fclose(file);
        return 1;
    }

    // Rewind file to read data
    rewind(file);

    // --- 2. Allocate Host (CPU) Memory ---
    char* h_encrypted = (char*)malloc(num_passwords * MAX_PW_LEN * sizeof(char));
    char* h_decrypted = (char*)malloc(num_passwords * MAX_PW_LEN * sizeof(char));

    if (!h_encrypted || !h_decrypted) {
        printf("Error: CPU memory allocation failed.\n");
        fclose(file);
        return 1;
    }

    // Read data into host memory (flat 1D array simulating 2D)
    int count = 0;
    while (fgets(buffer, MAX_PW_LEN, file) && count < num_passwords) {
        // Remove trailing newline
        size_t len = strlen(buffer);
        if (len > 0 && (buffer[len-1] == '\n' || buffer[len-1] == '\r')) {
            buffer[len-1] = '\0';
        }
        strncpy(h_encrypted + (count * MAX_PW_LEN), buffer, MAX_PW_LEN);
        count++;
    }
    fclose(file);

    // --- 3. Allocate Device (GPU) Memory ---
    char *d_encrypted, *d_decrypted;
    cudaMalloc((void**)&d_encrypted, num_passwords * MAX_PW_LEN * sizeof(char));
    cudaCheckError();

    cudaMalloc((void**)&d_decrypted, num_passwords * MAX_PW_LEN * sizeof(char));
    cudaCheckError();

    // Copy data from Host to Device
    cudaMemcpy(d_encrypted, h_encrypted, num_passwords * MAX_PW_LEN * sizeof(char), cudaMemcpyHostToDevice);
    cudaCheckError();

    // --- 4. Dynamically Split Blocks and Threads ---
    // The assignment asks for a dynamic split in the x dimension.
    int threads_per_block = 256; // Standard optimal thread count
    int blocks_per_grid = (num_passwords + threads_per_block - 1) / threads_per_block;

    printf("Processing %d passwords using %d blocks and %d threads per block...\n", 
            num_passwords, blocks_per_grid, threads_per_block);

    // Execute Kernel
    decrypt_kernel<<<blocks_per_grid, threads_per_block>>>(d_encrypted, d_decrypted, num_passwords);
    cudaDeviceSynchronize(); // Wait for GPU to finish
    cudaCheckError();

    // --- 5. Copy Results back to Host ---
    cudaMemcpy(h_decrypted, d_decrypted, num_passwords * MAX_PW_LEN * sizeof(char), cudaMemcpyDeviceToHost);
    cudaCheckError();

    // --- 6. Write to Output File ---
    // Make sure results directory exists
    system("mkdir -p results");
    FILE* out_file = fopen("results/decrypted.txt", "w");
    if (!out_file) {
        out_file = fopen("decrypted.txt", "w");
    }

    if (out_file) {
        for (int i = 0; i < num_passwords; i++) {
            fprintf(out_file, "%s\n", h_decrypted + (i * MAX_PW_LEN));
        }
        fclose(out_file);
        printf("Success: Decrypted passwords written to results/decrypted.txt\n");
    } else {
        printf("Error: Could not open output file.\n");
    }

    // --- 7. Memory Deallocation ---
    cudaFree(d_encrypted);
    cudaFree(d_decrypted);
    free(h_encrypted);
    free(h_decrypted);

    return 0;
}
EOF

# 3. Compile the CUDA program
echo "Compiling CUDA code..."
# Use nvcc to compile .cu files
nvcc main.cu -o main

# 4. Create results directory
mkdir -p results

# 5. Run the program with the sample passwords
echo "Running..."
./main encrypted_passwords.txt

# 6. Display the results
echo -e "\n--- OUTPUT ---"
cat results/decrypted.txt || cat decrypted.txt
