%%bash

# 1. Create a sample matrix input file
cat << 'EOF' > sample_matrix.txt
2 2
1.5 2.5
3.5 4.5
2 2
5.0 6.0
7.0 8.0
EOF

# 2. Write your OpenMP C code to main.c
cat << 'EOF' > main.c
#include <stdio.h>
#include <stdlib.h>
#include <omp.h>
#include <math.h>

typedef struct {
    int rows;
    int cols;
    double *data;
} Matrix;

// Helper to free a matrix
void free_matrix(Matrix *m) {
    if (m->data != NULL) {
        free(m->data);
        m->data = NULL;
    }
    m->rows = 0;
    m->cols = 0;
}

// Function to read a single matrix from file
// Expected format: ROWS COLS \n then ROWS*COLS elements
int read_matrix(FILE *file, Matrix *m) {
    if (fscanf(file, "%d %d", &m->rows, &m->cols) != 2) {
        return 0; // EOF or invalid header
    }
    
    if (m->rows <= 0 || m->cols <= 0) {
        printf("Error: Invalid matrix dimensions (%d x %d).\n", m->rows, m->cols);
        return -1;
    }

    m->data = (double *)malloc(m->rows * m->cols * sizeof(double));
    if (m->data == NULL) {
        printf("Error: Memory allocation failed for matrix.\n");
        return -1;
    }

    for (int i = 0; i < m->rows * m->cols; i++) {
        if (fscanf(file, "%lf", &m->data[i]) != 1) {
            printf("Error: Non-numeric value or missing elements in matrix data.\n");
            free_matrix(m);
            return -1;
        }
    }

    return 1; // Success
}

// Write matrix to file
void print_matrix(FILE *out, const char* name, int rows, int cols, double* data) {
    fprintf(out, "%s - %d,%d\n", name, rows, cols);
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (isnan(data[i * cols + j])) {
                fprintf(out, "NaN");
            } else {
                fprintf(out, "%.2f", data[i * cols + j]);
            }
            if (j < cols - 1) fprintf(out, ", ");
        }
        fprintf(out, "\n");
    }
    fprintf(out, "\n");
}

int main(int argc, char *argv[]) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <num_threads>\n", argv[0]);
        return 1;
    }

    const char *file_path = argv[1];
    int req_threads = atoi(argv[2]);

    if (req_threads <= 0) {
        printf("Error: Invalid number of threads.\n");
        return 1;
    }

    FILE *file = fopen(file_path, "r");
    if (!file) {
        printf("Error: Missing or cannot open input file '%s'\n", file_path);
        return 1;
    }

    FILE *out = fopen("results/results.txt", "w");
    if (!out) {
        out = fopen("results.txt", "w");
        if (!out) {
            printf("Error: Cannot open output results file.\n");
            fclose(file);
            return 1;
        }
    }

    int pair_count = 1;
    Matrix A, B;

    while (1) {
        int status_A = read_matrix(file, &A);
        if (status_A == 0) break; // EOF
        if (status_A == -1) goto cleanup;

        int status_B = read_matrix(file, &B);
        if (status_B == 0) {
            printf("Error: Missing matrix pair. File must contain an even number of matrices.\n");
            free_matrix(&A);
            goto cleanup;
        }
        if (status_B == -1) {
            free_matrix(&A);
            goto cleanup;
        }

        fprintf(out, "==== Pair %d ====\n\n", pair_count++);

        // Element-wise addition, subtraction, multiplication, division
        if (A.rows == B.rows && A.cols == B.cols) {
            int max_dim = A.rows * A.cols;
            int num_threads = (req_threads > max_dim) ? max_dim : req_threads;
            omp_set_num_threads(num_threads);

            double *add = (double *)malloc(max_dim * sizeof(double));
            double *sub = (double *)malloc(max_dim * sizeof(double));
            double *mul = (double *)malloc(max_dim * sizeof(double));
            double *div = (double *)malloc(max_dim * sizeof(double));

            #pragma omp parallel for
            for (int i = 0; i < max_dim; i++) {
                add[i] = A.data[i] + B.data[i];
                sub[i] = A.data[i] - B.data[i];
                mul[i] = A.data[i] * B.data[i];
                if (B.data[i] == 0.0) {
                    div[i] = NAN;
                } else {
                    div[i] = A.data[i] / B.data[i];
                }
            }

            print_matrix(out, "Addition", A.rows, A.cols, add);
            print_matrix(out, "Subtraction", A.rows, A.cols, sub);
            print_matrix(out, "Element-wise Multiplication", A.rows, A.cols, mul);
            print_matrix(out, "Element-wise Division", A.rows, A.cols, div);

            free(add);
            free(sub);
            free(mul);
            free(div);
        } else {
            fprintf(out, "Addition cannot be done (shapes differ).\n");
            fprintf(out, "Subtraction cannot be done (shapes differ).\n");
            fprintf(out, "Element-wise Multiplication cannot be done (shapes differ).\n");
            fprintf(out, "Element-wise Division cannot be done (shapes differ).\n\n");
        }

        // Transpose A
        int max_dim_A = A.rows * A.cols;
        int num_threads_A = (req_threads > max_dim_A) ? max_dim_A : req_threads;
        omp_set_num_threads(num_threads_A);
        
        double *transA = (double *)malloc(max_dim_A * sizeof(double));
        #pragma omp parallel for collapse(2)
        for (int i = 0; i < A.rows; i++) {
            for (int j = 0; j < A.cols; j++) {
                transA[j * A.rows + i] = A.data[i * A.cols + j];
            }
        }
        print_matrix(out, "Transpose of A", A.cols, A.rows, transA);
        free(transA);

        // Transpose B
        int max_dim_B = B.rows * B.cols;
        int num_threads_B = (req_threads > max_dim_B) ? max_dim_B : req_threads;
        omp_set_num_threads(num_threads_B);
        
        double *transB = (double *)malloc(max_dim_B * sizeof(double));
        #pragma omp parallel for collapse(2)
        for (int i = 0; i < B.rows; i++) {
            for (int j = 0; j < B.cols; j++) {
                transB[j * B.rows + i] = B.data[i * B.cols + j];
            }
        }
        print_matrix(out, "Transpose of B", B.cols, B.rows, transB);
        free(transB);

        // Matrix Multiplication
        if (A.cols == B.rows) {
            int max_dim_mul = A.rows * B.cols;
            int num_threads_mul = (req_threads > max_dim_mul) ? max_dim_mul : req_threads;
            omp_set_num_threads(num_threads_mul);

            double *matmul = (double *)malloc(max_dim_mul * sizeof(double));

            #pragma omp parallel for collapse(2)
            for (int i = 0; i < A.rows; i++) {
                for (int j = 0; j < B.cols; j++) {
                    double sum = 0.0;
                    for (int k = 0; k < A.cols; k++) {
                        sum += A.data[i * A.cols + k] * B.data[k * B.cols + j];
                    }
                    matmul[i * B.cols + j] = sum;
                }
            }
            print_matrix(out, "Matrix Multiplication", A.rows, B.cols, matmul);
            free(matmul);
        } else {
            fprintf(out, "Matrix Multiplication cannot be done (A.cols != B.rows).\n\n");
        }

        free_matrix(&A);
        free_matrix(&B);
        fprintf(out, "--------------------------\n");
    }

cleanup:
    fclose(file);
    fclose(out);
    printf("Processing complete. Results saved to results/results.txt\n");

    return 0;
}
EOF

# 3. Compile the OpenMP C program
echo "Compiling OpenMP code..."
# We must use -fopenmp for OpenMP and -lm for math.h
gcc main.c -o main -fopenmp -lm

# 4. Create results directory
mkdir -p results

# 5. Run the program with the sample matrix and 4 threads
echo "Running..."
./main sample_matrix.txt 4

# 6. Display the results
echo -e "\n--- OUTPUT ---"
cat results/results.txt || cat results.txt
